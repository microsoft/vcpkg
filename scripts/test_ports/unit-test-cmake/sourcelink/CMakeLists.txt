cmake_minimum_required(VERSION 3.19)
project(sourcelink-test C)

if(NOT MSVC OR CMAKE_C_COMPILER_VERSION VERSION_LESS "19.15.26726.0")
    return()
endif()

set(scan_dir "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/sourcelink")
set(VCPKG_SOURCELINK_FILE "${CMAKE_CURRENT_BINARY_DIR}/current port.json")
set(rollup "${CMAKE_BINARY_DIR}/CMakeFiles/vcpkg.sourcelink.json")
set(dependency_version "${TEST_MODE}")
set(has_transitive OFF)
file(REMOVE "${scan_dir}/transitive.json")
if(TEST_MODE STREQUAL "empty")
    file(REMOVE "${scan_dir}/dependency.json" "${VCPKG_SOURCELINK_FILE}")
    unset(VCPKG_SOURCELINK_FILE)
else()
    file(WRITE "${scan_dir}/dependency.json"
        "{\"documents\":{\"__VCPKG_INSTALLED_TRIPLET_DIR__/include/sample/*\":\"https://example.invalid/${TEST_MODE}/*;raw=1\"}}")
    file(WRITE "${VCPKG_SOURCELINK_FILE}"
        "{\"documents\":{\"current/*\":\"https://example.invalid/current/*\"}}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.c" "int main(void) { return 0; }\n")

function(check_sourcelink target)
    get_target_property(options "${target}" LINK_OPTIONS)
    if(TEST_MODE STREQUAL "empty")
        if(options)
            message(FATAL_ERROR "Stale SourceLink options on ${target}: ${options}")
        endif()
    else()
        if(NOT "/sourcelink:${rollup}" IN_LIST options)
            message(FATAL_ERROR "Missing SourceLink options on ${target}: ${options}")
        endif()
        file(READ "${rollup}" json)
        file(TO_NATIVE_PATH "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}" installed)
        string(JSON value GET "${json}" documents "${installed}/include/sample/*")
        if(NOT value STREQUAL "https://example.invalid/${dependency_version}/*;raw=1")
            message(FATAL_ERROR "Incorrect or stale SourceLink URL: ${value}")
        endif()
        string(JSON value GET "${json}" documents "current/*")
        if(NOT value STREQUAL "https://example.invalid/current/*")
            message(FATAL_ERROR "Missing current-port mapping")
        endif()
        if(has_transitive)
            string(JSON value GET "${json}" documents "${installed}/include/transitive/*")
            if(NOT value STREQUAL "https://example.invalid/transitive/*")
                message(FATAL_ERROR "Missing transitive dependency mapping")
            endif()
        endif()
    endif()
endfunction()

add_executable(first "${CMAKE_CURRENT_BINARY_DIR}/main.c")
check_sourcelink(first)

# Updated contents and newly installed dependencies must be picked up in the same configure.
if(NOT TEST_MODE STREQUAL "empty")
    set(dependency_version "${TEST_MODE}-updated")
    file(WRITE "${scan_dir}/dependency.json"
        "{\"documents\":{\"__VCPKG_INSTALLED_TRIPLET_DIR__/include/sample/*\":\"https://example.invalid/${dependency_version}/*;raw=1\"}}")
endif()
add_subdirectory(subdir)
if(NOT TEST_MODE STREQUAL "empty")
    set(has_transitive ON)
    file(WRITE "${scan_dir}/transitive.json"
        "{\"documents\":{\"__VCPKG_INSTALLED_TRIPLET_DIR__/include/transitive/*\":\"https://example.invalid/transitive/*\"}}")
endif()
add_executable(last "${CMAKE_CURRENT_BINARY_DIR}/main.c")
check_sourcelink(last)

if(NOT TEST_MODE STREQUAL "empty")
    file(TIMESTAMP "${rollup}" previous_timestamp "%s" UTC)
    execute_process(COMMAND "${CMAKE_COMMAND}" -E sleep 1.1 COMMAND_ERROR_IS_FATAL ANY)
    add_executable(unchanged "${CMAKE_CURRENT_BINARY_DIR}/main.c")
    check_sourcelink(unchanged)
    file(TIMESTAMP "${rollup}" current_timestamp "%s" UTC)
    if(NOT current_timestamp STREQUAL previous_timestamp)
        message(FATAL_ERROR "Unchanged SourceLink output was rewritten")
    endif()
endif()
