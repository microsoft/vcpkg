cmake_minimum_required(VERSION 3.20)
project(protobuf_consumer_tests LANGUAGES CXX)

option(PROTOBUF_TEST_FULL "Require and test the full runtime" OFF)
option(PROTOBUF_TEST_LIBPROTOC "Require and test libprotoc" OFF)
option(PROTOBUF_TEST_ZLIB "Require and test gzip streams" OFF)

find_package(protobuf CONFIG REQUIRED)
if(NOT TARGET protobuf::libprotobuf-lite OR NOT TARGET protobuf::protoc)
    message(FATAL_ERROR "Missing lite runtime or host protoc target")
endif()

# Both spellings must support CONFIG consumers, including lite installations.
find_package(Protobuf CONFIG REQUIRED)

enable_testing()
add_executable(lite_roundtrip lite_roundtrip.cpp lite.proto)
target_compile_features(lite_roundtrip PRIVATE cxx_std_17)
target_include_directories(lite_roundtrip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(lite_roundtrip PRIVATE protobuf::libprotobuf-lite)
protobuf_generate(TARGET lite_roundtrip)
add_test(NAME lite_roundtrip COMMAND lite_roundtrip)

if(PROTOBUF_TEST_FULL OR PROTOBUF_TEST_LIBPROTOC OR PROTOBUF_TEST_ZLIB)
    if(NOT TARGET protobuf::libprotobuf)
        message(FATAL_ERROR "The requested feature did not provide the full runtime")
    endif()
    set(protobuf_MODULE_COMPATIBLE ON)
    find_package(Protobuf CONFIG REQUIRED)
    if(NOT Protobuf_LIBRARIES OR NOT Protobuf_INCLUDE_DIRS)
        message(FATAL_ERROR "Full runtime module compatibility variables are missing")
    endif()
    add_executable(full_roundtrip full_roundtrip.cpp full.proto)
    target_compile_features(full_roundtrip PRIVATE cxx_std_17)
    target_include_directories(full_roundtrip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(full_roundtrip PRIVATE protobuf::libprotobuf)
    protobuf_generate(TARGET full_roundtrip)
    add_test(NAME full_roundtrip COMMAND full_roundtrip)
endif()

if(PROTOBUF_TEST_LIBPROTOC OR TARGET protobuf::libprotoc)
    add_executable(compiler_import compiler_import.cpp)
    target_compile_features(compiler_import PRIVATE cxx_std_17)
    target_link_libraries(compiler_import PRIVATE protobuf::libprotoc protobuf::libprotobuf)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/compiler-output")
    add_test(NAME compiler_import COMMAND compiler_import
        "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/compiler-output")
endif()

if(PROTOBUF_TEST_ZLIB)
    add_executable(gzip_roundtrip gzip_roundtrip.cpp)
    target_compile_features(gzip_roundtrip PRIVATE cxx_std_17)
    target_link_libraries(gzip_roundtrip PRIVATE protobuf::libprotobuf)
    add_test(NAME gzip_roundtrip COMMAND gzip_roundtrip)
endif()
